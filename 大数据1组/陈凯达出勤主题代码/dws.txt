set hive.exec.dynamic.partition.mode=nonstrict
WITH stud as
(  
    SELECT 
        c.class_date as dateinfo,
        c.class_id,
        s.student_id,
        s.signin_time,
        t.morning_begin_time,
        t.afternoon_begin_time,
        t.evening_begin_time,
        t.morning_end_time,
        t.afternoon_end_time,
        t.evening_end_time,
        substr(c.class_date,1,4) as yearinfo,
        substr(c.class_date,1,7) as monthinfo,
        c.class_date as dayinfo
from
zxedu_ods.dim_course_table_upload_detail_dimen c
LEFT JOIN zxedu_ods.dim_class_studying_student_count_dimen st
on c.class_id = st.class_id
LEFT JOIN zxedu_ods.dim_class_time_dimen t
on c.class_id = t.class_id
LEFT JOIN zxedu_ods.fact_student_signin_ods s
on c.class_id = s.class_id and c.class_date = s.signin_date
LEFT JOIN
zxedu_ods.dim_calendar d
on c.class_date = d.datelist
)
INSERT overwrite table zxedu_dws.student_attendance_dwm PARTITION (yearinfo,monthinfo,dayinfo)
select 
    dayinfo as dateinfo,
    class_id,
    student_id,
    if(
               --上午正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
               sum(
                   case
                   --上午打卡时间是否在上课前40分钟~下课时间段之内
                       when substr(signin_time,11,9) BETWEEN substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',morning_begin_time))-2400),11,9) and 
            morning_end_time
                           then 1   --上午来了
                       else 0 end   --上午没来
                ) > 0,  --打卡多次，只要有一次正常打卡，就会>0，返回true；否则没来，返回false
               if(
                   sum(
                       case
                   --上午打卡时间是否在上课前40分钟~上课后10分钟之内
                          when substr(signin_time,11,9) between substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',morning_begin_time))-2400),11,9)
                          AND substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',morning_begin_time))+600),11,9)
                              then 1    --正常出勤
                          else 0 end    --迟到
                    ) > 0,              --有一次打卡是正常出勤，就会>0，返回true；否则迟到，返回false
                   0,   --正常出勤
                   1    --迟到
                ),
               2    --上午没来
            ) as morning_att,
    if(
               --下午正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
               sum(
                   case
                   --下午打卡时间是否在上课前40分钟~下课时间段之内
                       when substr(signin_time,11,9) BETWEEN substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',afternoon_begin_time))-2400),11,9) and 
            afternoon_end_time
                           then 1   --下午来了
                       else 0 end   --下午没来
                ) > 0,  --打卡多次，只要有一次正常打卡，就会>0，返回true；否则没来，返回false
               if(
                   sum(
                       case
                   --下午打卡时间是否在上课前40分钟~上课后10分钟之内
                          when substr(signin_time,11,9) between substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',afternoon_begin_time))-2400),11,9)
                          AND substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',afternoon_begin_time))+600),11,9)
                              then 1    --正常出勤
                          else 0 end    --迟到
                    ) > 0,              --有一次打卡是正常出勤，就会>0，返回true；否则迟到，返回false
                   0,   --正常出勤
                   1    --迟到
                ),
               2    --下午没来
            ) as afternoon_att,
    if(
               --晚上正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
               sum(
                   case
                   --晚上打卡时间是否在上课前40分钟~下课时间段之内
                       when substr(signin_time,11,9) BETWEEN substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',evening_begin_time))-2400),11,9) and 
            evening_begin_time
                           then 1   --晚上来了
                       else 0 end   --晚上没来
                ) > 0,  --打卡多次，只要有一次正常打卡，就会>0，返回true；否则没来，返回false
               if(
                   sum(
                       case
                   --晚上打卡时间是否在上课前40分钟~上课后10分钟之内
                          when substr(signin_time,11,9) between substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',evening_begin_time))-2400),11,9)
                          AND substr(from_unixtime(unix_timestamp(concat('1970-01-01 ',evening_begin_time))+600),11,9)
                              then 1    --正常出勤
                          else 0 end    --迟到
                    ) > 0,              --有一次打卡是正常出勤，就会>0，返回true；否则迟到，返回false
                   0,   --正常出勤
                   1    --迟到
                ),
               2    --晚上没来
            ) as evening_att,
    yearinfo,
    monthinfo,
    dayinfo
from stud
GROUP BY dayinfo,monthinfo,yearinfo,class_id,student_id
having student_id is not null;



INSERT overwrite table zxedu_dws.class_attendance_dwm PARTITION (yearinfo,monthinfo,dayinfo)
SELECT
dayinfo as dateinfo,
class_id,
sum(if(morning_att='0',1,0)) as morning_att_count,
sum(if(afternoon_att='0',1,0)) as afternoon_att_count,
sum(if(evening_att='0',1,0)) as evening_att_count,
sum(if(morning_att='1',1,0)) as morning_late_count,
sum(if(afternoon_att='1',1,0)) as afternoon_late_count,
sum(if(evening_att='1',1,0)) as evening_late_count,
yearinfo,
monthinfo,
dayinfo
FROM 
zxedu_dws.student_attendance_dwm
GROUP BY dayinfo,monthinfo,yearinfo,class_id;



with stud as (
select
c.class_date,
c.class_id,
s.studying_date,
s.studying_student_count,
ca.morning_att_count,
l.morning_leave_count,
ca.afternoon_att_count,
l.afternoon_leave_count,
ca.evening_att_count,
l.evening_leave_count,
year(c.class_date) as yearinfo,
substr(c.class_date,1,7) as monthinfo,
c.class_date as dayinfo
from
zxedu_ods.dim_course_table_upload_detail_dimen c
left join
zxedu_ods.dim_class_studying_student_count_dimen s
on c.class_id = s.class_id
left join
zxedu_dws.class_attendance_dwm ca
on c.class_id = ca.class_id
left join
zxedu_dws.class_leave_dwm l
on c.class_id = l.class_id
)
--INSERT overwrite table zxedu_dws.class_truant_dwm PARTITION (yearinfo,monthinfo,dayinfo)
select
dayinfo as class_date,
class_id,
cast(sum(coalesce(cast(studying_student_count as int),0))-sum(coalesce(cast(morning_att_count as int),0))-sum(coalesce(cast(morning_leave_count as int),0)) as string) as morning_truant_count,
cast(sum(coalesce(cast(studying_student_count as int),0))-sum(coalesce(cast(afternoon_att_count as int),0))-sum(coalesce(cast(afternoon_leave_count as int),0)) as string) as afternoon_truant_count,
cast(sum(coalesce(cast(studying_student_count as int),0))-sum(coalesce(cast(evening_att_count as int),0))-sum(coalesce(cast(evening_leave_count as int),0)) as string) as evening_truant_count,
yearinfo,
monthinfo,
dayinfo
from 
stud
where 
class_date = studying_date
group by dayinfo,monthinfo,yearinfo,class_id;




INSERT overwrite table zxedu_dws.class_truant_dwm PARTITION (yearinfo,monthinfo,dayinfo)
select 
c.dateinfo,
st.class_id,
max(st.studying_student_count-cast(c.morning_att_count as int)-cast(coalesce(l.morning_leave_count,0)as int))as morning_truant_count,
max(st.studying_student_count-cast(c.afternoon_att_count as int)-cast(coalesce(l.morning_leave_count,0)as int))as afternoon_truant_count,
max(st.studying_student_count-cast(c.evening_att_count as int)-cast(coalesce(l.morning_leave_count,0)as int))as evening_truant_count,
c.yearinfo,
c.monthinfo,
c.dateinfo
from 
zxedu_ods.dim_class_studying_student_count_dimen st
left join zxedu_dws.class_attendance_dwm c
on c.class_id = st.class_id and c.dayinfo = st.studying_date
left join zxedu_dws.class_leave_dwm l
on st.class_id = l.class_id
where c.class_id is not null
group by c.dateinfo,c.monthinfo,c.yearinfo,st.class_id;
