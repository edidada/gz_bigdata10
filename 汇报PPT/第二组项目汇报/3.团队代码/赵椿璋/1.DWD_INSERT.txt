INSERT overwrite table oe_dwd.fact_customer_appeal partition (dt)
SELECT  id ,
        customer_relationship_first_id ,
        employee_id ,
        employee_name ,
        employee_department_id ,
        employee_tdepart_id ,
        case appeal_status
        when 0 then "待稽核"
        when 1 then "无效"
        when 2 then "有效"
        end as 
        appeal_status ,
        audit_id ,
        audit_name ,
        audit_department_id ,
        audit_department_name ,
        audit_date_time ,
        create_date_time ,
        update_date_time ,
        deleted ,
        tenant ,
    substr(create_date_time, 1, 10) dt
from oe_ods.customer_appeal
where customer_relationship_first_id IS NOT NULL;

INSERT overwrite table oe_dwd.fact_student_leave_apply partition (dt)
SELECT  id ,
        class_id ,
        student_id ,
        case  audit_state
        when 0 then "待审核"
        when 1 then "通过"
        when 2 then "不通过"
        end as  audit_state,
        audit_person ,
        audit_time ,
        audit_remark ,
        case leave_type
        WHEN 1 then "请假"
        WHEN 2 then "销假"
        end as leave_type,
        CASE leave_reason
        WHEN 1 then "事假"
        WHEN 2 then "病假"
        end as leave_reason,
        begin_time ,
        case begin_time_type
        when 1 then "上午"
        WHEN 2 THEN "下午"
        end as begin_time_type ,
        end_time ,
        case end_time_type
        WHEN 1 then "上午"
        WHEN 2 then "下午"
        end as 
        end_time_type ,
        days ,
        CASE cancel_state
        WHEN 0 then "未撤销"
        when 1 then "已撤销"
        end as 
        cancel_state ,
        cancel_time ,
        old_leave_id ,
        leave_remark ,
        CASE valid_state
        when 0 THEN "无效"
        WHEN 1 THEN "有效"
        end as valid_state,
        create_time ,
    substr(create_time, 1, 10) dt
from oe_ods.student_leave_apply
where class_id IS NOT NULL;
