CREATE table oe_dws.Attendance_theme(
    class_date STRING COMMENT '课程日期',
    class_id int COMMENT '班级id',
    student_id int COMMENT '学生id',
    all_people int COMMENT '总人数',
    morning_attendance_number int COMMENT '早上出勤人数',
    morning_lated_number int COMMENT '早上迟到人数',
    morning_leave_number int COMMENT '早上请假人数',
    afternoon_attendance_number int COMMENT '下午出勤人数',
    afternoon_lated_number int COMMENT '下午迟到人数',
    afternoon_leave_number int COMMENT '下午请假人数',
    everning_attendance_number int COMMENT '晚上出勤人数',
    everning_lated_number int COMMENT '晚上迟到人数',
    everning_leave_number int COMMENT '晚上请假人数',
    group_type STRING COMMENT '分组类型',
    the_year string COMMENT '年',
    the_month string COMMENT '月',
    the_day string COMMENT '日'
)
PARTITIONED BY (dt STRING)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY '\t'
stored as orcfile
TBLPROPERTIES ('orc.compress'='SNAPPY');
