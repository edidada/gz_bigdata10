--建表
drop table if exists itcast_dws.student_attendance;
create database if not exists itcast_dws;
create table itcast_dws.student_attendance
(
	class_id int,
	student_id int,
	signin_date longtext,
	signin_time longtext,
	signin_time_type longtext,
	morn_attend longtext comment '上午出勤状态:（正常出勤：0,迟到：1,其他：2）',
	aftn_attend longtext comment '下午出勤状态:（正常出勤：0,迟到：1,其他：2）',
	even_attend longtext comment '晚上出勤状态:（正常出勤：0,迟到：1,其他：2）'
)
comment '学生出勤（正常出勤和迟到）数据'
partitioned by(dt_info longtext)
row format delimited
fields terminated by '\t'
stored as orc
tblproperties ('orc.compress'='SNAPPY');

set hive.exec.max.dynamic.partitions.pernode=10000;

set hive.exec.max.created.files=150000;

--全量插入语句
--子查询1：出勤时间
with tmp1 as(
	select 
		sign.class_id,
		sign.signin_date,
		sign.student_id,
		unix_timestamp(concat(sign.signin_date,' ',sublongtext (sign.signin_time,12,8))) as signin_time,
		--正常出勤时间段
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.morning_begin_time)) -2400)
		as morning_begin_date_before40,	
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.morning_begin_time)) +600)
		as morning_begin_date_after10,	
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.afternoon_begin_time)) -2400)
		as afternoon_begin_date_before40,	
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.afternoon_begin_time)) +600)
		as afternoon_begin_date_after10,		
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.evening_begin_time)) -2400)
		as evening_begin_date_before40,		
		from_unixtime(unix_timestamp(concat(sign.signin_date,' ',timet.evening_begin_time)) +600)
		as evening_begin_date_after10,	
		--结束时间转换（迟到时间段）
		concat(sign.signin_date,' ',timet.morning_end_time)
		as morning_end_date,
		concat(sign.signin_date,' ',timet.afternoon_end_time)
		as afternoon_end_date,
		concat(sign.signin_date,' ',timet.evening_end_time)
		as evening_end_date
	--关联条件
	from itcast_ods.tbh_student_signin_record sign
		left join itcast_ods.course_table_upload_detail course
			on sign.class_id= course.class_id 
				and course.class_date = sign.signin_date
				
		left join itcast_ods.tbh_class_time_table timet
			on course.class_id= timet.class_id		
	where course.content != '开班典礼' 
		and course.class_date >= timet.use_begin_date 
			and course.class_date <= timet.use_end_date	
	--分组【执行顺序】
	--group by sign.class_id,sign.date
),
--子查询2筛选出有效签到时间，且分为上午下午晚自习
tmp2 as(
	select 
		*,
		case
			when signin_time between unix_timestamp(morning_begin_date_before40) and unix_timestamp(morning_end_date)
			then 'morning_signin_time'
			when signin_time between unix_timestamp(afternoon_begin_date_before40) and unix_timestamp(afternoon_end_date)
			then 'afternoon_signin_time'
			when signin_time between unix_timestamp(evening_begin_date_before40) and unix_timestamp(evening_end_date)
			then 'evening_signin_time'
			else null
		end
		as signin_time_type
		
	from tmp1
),
--子查询3：去重，只选择每个时间段内的第一条打卡数据
tmp3 as (
	select
		*,
	    row_number() over(partition by signin_time_type,student_id,signin_date order by signin_time ) rn
    from tmp2
	--筛选出有效打卡的数据
	where signin_time_type is not null
)
insert into table itcast_dws.student_attendance partition (dt_info)
select
	max(class_id),
	student_id,
	signin_date,
	max(
),
	max(signin_time_type),
	min(case
			when signin_time between unix_timestamp(morning_begin_date_before40) and unix_timestamp(morning_begin_date_after10) 
			then 0
			when signin_time between unix_timestamp(morning_begin_date_after10) and unix_timestamp(morning_end_date)
			then 1
			else 2
		end)
	as morn_attend,
	min(case
			when signin_time between unix_timestamp(afternoon_begin_date_before40) and unix_timestamp(afternoon_begin_date_after10)
			then 0
			when signin_time between unix_timestamp(afternoon_begin_date_after10) and unix_timestamp(afternoon_end_date)
			then 1
			else 2
		end)
	as aftn_attend,
	min(case
			when signin_time between unix_timestamp(evening_begin_date_before40) and unix_timestamp(evening_begin_date_after10)
			then 0
			when signin_time between unix_timestamp(evening_begin_date_after10) and unix_timestamp(evening_end_date)
			then 1
			else 2
		end)
	as even_attend,
	'2022-08-11 'as dt_info
from tmp3
--去重：筛选出每个时间段打卡的第一条记录
where rn =1
group by student_id,signin_date
;
--检验数据

--建表语句
create database if not exists itcast_dws;
create table itcast_dws.class_attendance
(
	class_id int,
	signin_date longtext,
	
	--班级出勤
	class_morn_attend longtext comment '上午出勤人数',
	calss_aftn_attend longtext comment '下午出勤人数',
	class_even_attend longtext comment '晚自习出勤人数',
	--班级迟到
	class_morn_late longtext comment '上午迟到人数',
	calss_aftn_late longtext comment '下午迟到人数',
	class_even_late longtext comment '晚自习迟到人数'
)
comment '班级请假数据'
partitioned by(dt_info longtext)
row format delimited
fields terminated by '\t'
stored as orc
tblproperties ('orc.compress'='SNAPPY');

--全量插入语句
insert into itcast_dws.class_attendance partition (dt_info)
select
    attend.class_id,
    attend.signin_date,
	sum(if(attend.morn_attend = 0 or attend.morn_attend = 1,1,0))as class_morn_attend,
	sum(if(attend.aftn_attend = 0 or attend.morn_attend = 1,1,0))as class_aftn_attend,
	sum(if(attend.aftn_attend = 0 or attend.morn_attend = 1,1,0))as class_even_attend,
	sum(if(attend.morn_attend = 1,1,0)) as class_morn_late,
	sum(if(attend.aftn_attend = 1,1,0)) as class_aftn_late,
	sum(if(attend.even_attend = 1,1,0)) as class_even_late,
	'2022-08-12' as dt_info
from  itcast_dws.student_attendance attend
group by attend.class_id, attend.signin_date;

--建表语句
drop table if exists itcast_dws.class_leave_apply;
create table itcast_dws.class_leave_apply
(
	class_id int,
	signin_date longtext,
	--班级请假
	class_morn_apply longtext comment '上午请假人数',
	class_aftn_apply longtext comment '下午请假人数',
	class_even_apply longtext comment '晚自习请假人数'
)
comment '班级请假数据'
partitioned by(dt_info longtext)
row format delimited
fields terminated by '\t'
stored as orc
tblproperties ('orc.compress'='SNAPPY');
--涉及多个distinct去重，关闭数据倾斜优化
set hive.groupby.skewindata=false;

--全量插入语句
with tmp1 as (
select 
    apply.class_id,
    apply.student_id,
    course.class_date,
    unix_timestamp(concat(course.class_date,' ',timet.morning_begin_time)) as morning_begin_time,
    unix_timestamp(concat(course.class_date,' ',timet.afternoon_begin_time)) as afternoon_begin_time,
    unix_timestamp(concat(course.class_date,' ',timet.evening_begin_time)) as evening_begin_time,
    unix_timestamp(sublongtext(apply.begin_time,1,19)) as begin_time,
    unix_timestamp(sublongtext(apply.end_time,1,19)) as end_time 
from itcast_ods.student_leave_apply apply
	left join itcast_ods.course_table_upload_detail course
		on course.class_id = apply.class_id
	left join itcast_ods.tbh_class_time_table timet
		on apply.class_id = timet.class_id
where course.content != '开班典礼' and course.content!=''
	and course.class_date between timet.use_begin_date and timet.use_end_date
		and apply.audit_state = 1 and cancel_state = 0 and valid_state = 1
)
insert into itcast_dws.class_leave_apply partition (dt_info)
select
    class_id,
    class_date,
	count(distinct
	    case 
    		when morning_begin_time between begin_time and end_time
    		then student_id
    		else null
	    end)
	    as morn_apply,
	count(distinct
	    case 
    		when afternoon_begin_time between begin_time and end_time
    		then  student_id
    		else null
	    end)
	    as aftn_apply,
	    count(distinct
	    case 
    		when evening_begin_time between begin_time and end_time
    		then student_id
    		else null
	    end)
	    as even_apply,
	    '2022-08-11' as dt_info
from tmp1 
group by class_id,class_date
;

--建表语句
drop table if EXISTS itcast_dws.class_truant;
CREATE TABLE IF NOT EXISTS itcast_dws.class_truant (
    dateinfo      String        comment '日期',
    class_id        int           comment '班级id',
    morning_truant_count       String        comment '上午旷课人数',
    afternoon_truant_count   String        comment '下午旷课人数',
    evening_truant_count       String        comment '晚自习旷课人数')
comment '班级请假数据统计'
PARTITIONED BY (dt_info STRING)
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY '\t'
stored as orc
TBLPROPERTIES ('orc.compress'='SNAPPY');
--全量插入数据语句
insert into table itcast_dws.class_truant partition (dt_info)
SELECT 
    countt.studying_date as dateinfo,
    countt.class_id,
	(countt.studying_student_count - attend.class_morn_attend -  nvl(apply.class_morn_apply ,0))
    as morning_truant_count,
    (countt.studying_student_count - attend.calss_aftn_attend  - nvl(apply.class_aftn_apply,0))
    as afternoon_truant_count,
    (countt.studying_student_count -  attend.class_even_attend  -  nvl(apply.class_even_apply,0))
    as evening_truant_count,
    '2022-08-11' as dt_info
from itcast_dws.class_leave_apply apply
    left join itcast_ods.class_studying_student_count countt
        on apply.class_id = countt.class_id 
            and apply.signin_date = countt.studying_date
    left join  itcast_dws.class_attendance attend
        on attend.class_id = apply.class_id
            and attend.signin_date = apply.signin_date
;
select * from itcast_dws.class_truant;
