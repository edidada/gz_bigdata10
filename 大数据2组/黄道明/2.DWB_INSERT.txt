insert into oe_dwb.scrm_daoming partition(dt)
select
    cr.customer_id,
    cr.itcast_school_id ,
    ic.`itcast_school_name` ,
    ic. `itcast_subject_id` ,
    ic. `itcast_subject_name` ,
    ee.`tdepart_id` ,
    ee.`department_name` ,
    cr.`origin_type` ,
    if(cr.origin_type='NETSERVICE','线上','线下')as origin_type_stat,   
    
    substr(cr.payment_time,1,10) as payment_time,
    '2022-3-12' as dt
  
    from oe_dwd.fact_customer_relationship cr left join oe_dwd.dim_itcast_clazz ic on cr.itcast_clazz_id = ic.id
    left join oe_dwd.fact_customer_appeal ca on cr.id = ca.customer_relationship_first_id 
    left join oe_dwd.dim_employee ee on cr.creator = ee.id 
    left join oe_dwd.dim_scrm_department sd on ee.tdepart_id = sd.id
    left join oe_dwd.fact_customer_clue cc on cr.id = cc.customer_relationship_id;
