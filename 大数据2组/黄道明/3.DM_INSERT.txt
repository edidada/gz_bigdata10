-- 需求5:意向用户报名转化率,新增的意向客户中报名的客户占比。全部报名人数 / 全部新增的意向人数
union all
SELECT  
     
   round((signup_num.num / total_num.num * 100),2) ,  
   null as itcast_school_name,
   null as itcast_subject_name,
   null as tdepart_name,
   null as origin_type,
   null as payment_time_year ,
   null as payment_time_month ,
   null as payment_time_day,
   null as create_date_year ,
   null as create_date_month ,
   total_num.date_day ,
  
   'all' as groupType ,
   'all' as time_type  ,
   'PAID' as payment_state,    '2022-3-12' as dt
   
FROM  
   (  
       SELECT  
            count( 1 ) AS num,  
            DATE_FORMAT( cr.create_date_time, 'y-M-d' ) AS date_day   
        FROM  
            oe_ods.customer_relationship cr   
        GROUP BY  
            DATE_FORMAT( cr.create_date_time, 'y-M-d' )   
    ) total_num  
    INNER JOIN (  
        SELECT  
            count( 1 ) AS num,  
            DATE_FORMAT( cr.create_date_time, 'y-M-d' ) AS date_day   
        FROM  
            oe_ods.customer_relationship cr   
        WHERE  
            cr.payment_state = 'PAID'   
        GROUP BY  
            DATE_FORMAT( cr.create_date_time, 'y-M-d' )   
    ) signup_num ON total_num.date_day = signup_num.date_day 
union all    
-- 需求6:有效线索报名转化率,线上报名量 / 线上有效线索量
-- insert into table 
-- 按天

SELECT  
  
  
   
   round((signup_num.num / total_num.num * 100),2) ,  
   null as itcast_school_name,
   null as itcast_subject_name,
   null as tdepart_name,
   null as origin_type,
   null as payment_time_year ,
   null as payment_time_month ,
   null as payment_time_day ,
   null as create_date_year,
  null as create_date_month ,
   total_num.date_day ,
   
  
   'all' as groupType ,
   'year' as time_type  ,
   'PAID' as payment_state,    '2022-3-12' as dt
FROM 
   (  
   SELECT  
        count( 1 ) AS num,  
        DATE_FORMAT( cr.create_date_time, 'y-M-D' ) AS date_day   
    
   FROM  
        oe_ods.customer_clue cc  
        LEFT JOIN oe_ods.customer_relationship cr ON cr.id = cc.customer_relationship_id  
     WHERE  
        cc.clue_state IN ( 'VALID_NEW_CLUES', 'VALID_PUBLIC_NEW_CLUE' ) 
         AND cc.deleted ='0'   
        AND cr.id NOT IN (SELECT customer_relationship_first_id FROM oe_ods.customer_appeall WHERE appeal_status = '1' )   
    GROUP BY  
        DATE_FORMAT( cr.create_date_time, 'y-M-D')   
    ) total_num  
    INNER JOIN (  
    SELECT  
        count( 1 ) AS num,  
        DATE_FORMAT( cr.create_date_time, 'y-M-D' ) AS date_day   
    FROM  
        oe_ods.customer_relationship cr   
    WHERE  
        cr.payment_state = 'PAID'   
    GROUP BY  
        DATE_FORMAT( cr.create_date_time, 'y-M-D' )   
    ) as signup_num ON total_num.date_day = signup_num.date_day

     count( 1 ) ,  
   null as itcast_school_name,

    ,sd.`name`,
   null as payment_time_year ,
   null as payment_time_month ,
   null as payment_time_day,
   null as create_date_year ,
   null as create_date_month ,
   null as create_date_day ,

   'all' as time_type  ,
   'PAID' as payment_state,    '2022-3-12' as dt

    oe_ods.customer_relationship cr  
    LEFT JOIN oe_ods.employee e ON cr.creator = e.id  
    LEFT JOIN oe_ods.scrm_department sd ON e.tdepart_id = sd.id   
WHERE  
    cr.payment_state = 'PAID'   
    -- AND cr.payment_time >= '2019-10-01'
