with temp as(
    select
        class_id,
        class_date,
        --总人数
        tea.studying_student_count,
        --学生id
        tea.t5_student_id,
        --签到时间
        signin_time,
        --请假信息
        student_id,
        tea.begin_time,
        tea.end_time,
        --请假人数的去重
        row_number() over (partition by student_id order by tea.begin_time asc) student_id_one,
        
        
        
        --出勤人数的去重
        row_number() over (partition by t5_student_id order by signin_time asc) t5_student_id_one,
        --时间段
        case
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.morning_begin_time))-2400),12,8) and tea.morning_end_time) or student_id is not null
        then '早上出勤'
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.afternoon_begin_time))-2400),12,8) and tea.afternoon_end_time)
        then '下午出勤'
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.evening_begin_time))-2400),12,8) and tea.evening_end_time)
        then '晚上出勤'
        else '无效打卡时间'
        end
        as time_quantum,
        
        
        --打卡状态
        case
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.morning_begin_time))-2400),12,8) and substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.morning_begin_time))+600),12,8))
        then '早上正常出勤'
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.morning_begin_time))+600),12,8) and tea.morning_end_time)
        then '早上迟到出勤'
        
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.afternoon_begin_time))-2400),12,8) and substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.afternoon_begin_time))+600),12,8))
        then '下午正常出勤'
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.afternoon_begin_time))+600),12,8) and tea.afternoon_end_time)
        then '下午迟到出勤'
        
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.evening_begin_time))-2400),12,8) and substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.evening_begin_time))+600),12,8))
        then '晚上正常出勤'
        when (substring (signin_time,12,8) between substr(FROM_UNIXTIME(UNIX_TIMESTAMP(concat('1900-12-12 ',tea.evening_begin_time))+600),12,8) and tea.evening_end_time)
        then '晚上迟到出勤'
        
        
        end
        as clock_state,
    
    
        --当天
        tea.class_date the_day,
        substring (tea.class_date,1,7) the_month,
        substring (tea.class_date,1,4) the_year


from oe_dwb.teach_final tea
)

        else null

    --晚上请假率
    
    --晚上旷课人数
    
    --晚上旷课人数率
