INSERT INTO TABLE oe_dwd.fact_tbh_student_signin_record PARTITION (dt)
SELECT 
    id,
    
    CASE normal_class_flag
    WHEN 1 then '正课'
    WHEN 2 then '自习'
    end
    as normal_class_flag,
    
    time_table_id,
    class_id,
    student_id,
    signin_time,
    signin_date,
    CASE  inner_flag
    WHEN 0 then '外网'
    WHEN 1 then '内网'
    END
    as inner_flag,
    
    CASE signin_type
    WHEN 1 then '心跳打卡'
    WHEN 2 then '老师打卡'
    end
    as signin_type,
    
    case share_state
    WHEN 0 THEN '否'
    WHEN 1 THEN '是'
    end
    as share_state,
    inner_ip,
    signin_date create_time,
    dt
from oe_ods.tbh_student_signin_record;
