```hive
insert into mysql.itcast_dws.customer_signup_dws
SELECT
    payment_time,
    ic.itcast_school_name ,
    ic.itcast_subject_name ,
    case
        when grouping (ic.itcast_school_id)=0
        then '校区'
        when grouping (ic.itcast_subject_id)=0
        then '学科'
        when grouping (origin_type) = 0
        then '来源类型'
        else '总量'
        end

        as amount_type,
    count(1)
FROM hive.itcast_dwd.fact_customer_relationship cr
    LEFT JOIN hive.itcast_dwd.fact_itcast_clazz ic on cr.itcast_clazz_id=ic.id
WHERE cr.payment_state='PAID'

GROUP BY
    grouping sets
--校区
((cr.payment_time,ic.itcast_school_id,ic.itcast_school_name,cr.origin_type),
--学科
(cr.payment_time,ic.itcast_subject_id,ic.itcast_subject_name,cr.origin_type),
--来源类型
(cr.payment_time,origin_type),
--总量
()
);
```
