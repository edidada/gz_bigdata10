### 2、全量导入数据语句

#### （1）学生出勤信息表

```sql

-- 动态分区配置
set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nonstrict;
set hive.exec.max.dynamic.partitions.pernode=10000;
set hive.exec.max.dynamic.partitions=100000;
set hive.exec.max.created.files=150000;

-- hive压缩
set hive.exec.compress.intermediate=true;
set hive.exec.compress.output=true;
-- 写入时压缩生效
set hive.exec.orc.compression.strategy=COMPRESSION;

-- 写入数据强制分桶
set hive.enforce.bucketing=true;
-- 开启强制排序
set hive.enforce.sorting=true;

insert into itcast_dws.student_attendance_dwm partition(yearinfo, monthinfo, dayinfo)

select 
    date_time.dateinfo,
    ctud.class_id,
    tssr.student_id,
    -- 上午正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2 
    case
        when sum(
            case
                when substring(tssr.signin_time,12,8)
                    between from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') 
                    and tctt.morning_end_time
                then 1   -- 上午来了
                else 0 
                end   -- 上午没来
            ) > 0
        -- 打卡多次，只要有一次正常打卡，就会>0,否则没来
        then if(
            sum(
                case
                    when substring(tssr.signin_time,12,8) 
                        between from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') 
                        and from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss')
                    then 1    -- 正常出勤
                    else 0    -- 迟到
                end    
                ) > 0,0,1)
        else 2
        end as morning_att,
    -- 下午正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2    
    case 
        when sum(
            case
            when substring(tssr.signin_time,12,8) 
                between from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss')
                and tctt.afternoon_end_time
                then 1
                else 0 
            end) > 0
        
        then if(sum(
            case
            when substring(tssr.signin_time,12,8) 
                between from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss')
                and from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss') 
            then 1
            else 0 
            end) > 0,0,1)
        else 2
        end as afternoon_att,
       -- 晚自习正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
    case
        when sum(
            case
                when substring(tssr.signin_time,12,8)
                    between from_unixtime(unix_timestamp(tctt.evening_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') 
                    and tctt.evening_end_time
                then 1
                else 0 
            end) > 0
        then if(sum(
                case
                    when substring(tssr.signin_time,12,8)
                        between from_unixtime(unix_timestamp(evening_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') 
                        and from_unixtime(unix_timestamp(evening_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss')
                    then 1
                    else 0 
                end) > 0,0,1)
        else 2
        end as evening_att,
    substring(date_time.dateinfo,1,4) as yearinfo,
    substring(date_time.dateinfo,6,2) as monthinfo,
    substring(date_time.dateinfo,9,2) as dayinfo
    
  
from (select datelist as dateinfo from itcast_ods.calendar where datelist between '2019-09-01' and '2019-09-30') as date_time
         -- 日期课表不为空且不是开班典礼
        left join itcast_ods.course_table_upload_detail ctud on ctud.class_date = date_time.dateinfo 
            and coalesce(ctud.content,'') != '' 
            and ctud.content !='开学典礼'
    -- 学生打卡记录日期和班级匹配
        left join itcast_ods.tbh_student_signin_record tssr on tssr.class_id = ctud.class_id
            and tssr.signin_date = date_time.dateinfo 
    -- 获取班级作息时间以判断是否按时出勤
        left join itcast_ods.tbh_class_time_table tctt on tctt.id = tssr.time_table_id
    -- 按照日期、班级、学生分组统计
group by date_time.dateinfo, ctud.class_id, tssr.student_id,substring(date_time.dateinfo,1,4) ,substring(date_time.dateinfo,6,2) ,substring(date_time.dateinfo,9,2) ;
```

    satt.class_id,
    -- 上午出勤人数(包括迟到)
    count(distinct (case
         when satt.morning_att = 0 or satt.morning_att = 1 
         then satt.student_id
         else null end)) as morning_att_count,
    -- 下午出勤人数(包括迟到)
    count(distinct (case
        when satt.afternoon_att = 0 or satt.afternoon_att = 1
        then satt.student_id
        else null end)) as afternoon_att_count,
     --晚自习出勤人数(包括迟到)
    count(distinct (case
        when satt.evening_att = 0 or satt.evening_att = 1
        then satt.student_id
        else null end)) as evening_att_count,
      --上午迟到人数
    count(distinct(case
        when satt.morning_att = 1 
        then satt.student_id 
        else null end)) as morning_late_count,
        --下午迟到人数
    count(distinct(case 
        when satt.afternoon_att = 1 
        then satt.student_id 
        else null end)) as afternoon_late_count,
    --晚自习迟到人数
    count(distinct(case
        when satt.evening_att = 1 
        then satt.student_id 
        else null end))     as evening_late_count,
    
    substring(satt.dateinfo,1,4) as yearinfo,
    substring(satt.dateinfo,6,2) as monthinfo,
    substring(satt.dateinfo,9,2) as dayinfo
    
  
from itcast_dws.student_attendance_dwm satt  
     -- 获取班级作息时间以判断是否按时出勤
    left join itcast_ods.tbh_class_time_table tctt
    on tctt.class_id = satt.class_id 
    where satt.dateinfo >= tctt.use_begin_date and satt.dateinfo <= tctt.use_end_date
    -- 按照日期和班级统计
group by satt.dateinfo, satt.class_id,substring(satt.dateinfo,1,4) ,substring(satt.dateinfo,6,2) ,substring(satt.dateinfo,9,2);

#### (3) 班级请假信息表

```sql
-- 动态分区配置
set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nonstrict;
set hive.exec.max.dynamic.partitions.pernode=10000;
set hive.exec.max.dynamic.partitions=100000;
set hive.exec.max.created.files=150000;

-- hive压缩
set hive.exec.compress.intermediate=true;
set hive.exec.compress.output=true;
-- 写入时压缩生效
set hive.exec.orc.compression.strategy=COMPRESSION;

-- 写入数据强制分桶
set hive.enforce.bucketing=true;
-- 开启强制排序
set hive.enforce.sorting=true;

insert into itcast_dws.class_leave_dwm partition(yearinfo, monthinfo, dayinfo)

select 
    cud.class_date as dateinfo,
    cud.class_id,
    -- 请假结束时间 >= 早上课程开始时间 >= 请假开始时间，认为上午请假了
    count(distinct 
        if((concat(cud.class_date, ' ', ct.morning_begin_time) >= sla.begin_time
        and concat(cud.class_date, ' ', ct.morning_begin_time) <= sla.end_time),
        sla.student_id,0)) 
    as morning_leave_count,
      -- 请假结束时间 >= 下午课程开始时间 >= 请假开始时间，认为下午请假了
    count(distinct 
        if((concat(cud.class_date, ' ', ct.afternoon_begin_time) >= sla.begin_time
        and concat(cud.class_date, ' ', ct.afternoon_begin_time) <= sla.end_time),
        sla.student_id,0)) 
    as afternoon_leave_count,
      -- 请假结束时间 >= 晚上自习开始时间 >= 请假开始时间，认为晚上上午请假了
    count(distinct 
        if((concat(cud.class_date, ' ', ct.evening_begin_time) >= sla.begin_time
        and concat(cud.class_date, ' ', ct.evening_begin_time) <= sla.end_time),
        sla.student_id,0)) 
    as evening_leave_count,
    substring(cud.class_date,1,4) as yearinfo,
    substring(cud.class_date,6,2) as monthinfo,
    substring(cud.class_date,9,2) as dayinfo
    
from itcast_ods.student_leave_apply sla left join 
     itcast_ods.tbh_class_time_table ct on sla.class_id =ct.class_id
     --   请假表，请假状态已审核通过，且没有取消、数据有效
    and sla.audit_state = 1
    and sla.cancel_state = 0
    and sla.valid_state = 1
    left join itcast_ods.course_table_upload_detail cud on ct.class_id=cud.class_id
--   课程表，当天有课程内容
    and cud.content is not null
    and cud.content != '开班典礼'

--   作息时间表，数据在生效期范围内
 where cud.class_date between ct.use_begin_date and ct.use_end_date
	group by cud.class_date, cud.class_id;
```



### (4)班级旷课信息表

```sql
-- 动态分区配置


set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nonstrict;
set hive.exec.max.dynamic.partitions.pernode=10000;
set hive.exec.max.dynamic.partitions=100000;
set hive.exec.max.created.files=150000;

-- hive压缩
set hive.exec.compress.intermediate=true;
set hive.exec.compress.output=true;
-- 写入时压缩生效
set hive.exec.orc.compression.strategy=COMPRESSION;

-- 写入数据强制分桶
set hive.enforce.bucketing=true;
-- 开启强制排序
set hive.enforce.sorting=true;

with clc as 
(
    select 
        css.studying_date as dateinfo,
        css.class_id as class_id,
        --判断用字段
        -- 课程总人数
        css.studying_student_count as studying_student_count, 
        --早上出勤人数
        cad.morning_att_count as morning_att_count, 
         --早上请假人数
        if(cld.morning_leave_count is not null,cld.evening_leave_count,0) 
    as morning_leave_count,
        -- 下午出勤人数
        cad.afternoon_att_count as afternoon_att_count, 
        --中午请假人数
        if(cld.afternoon_leave_count is not null,cld.afternoon_leave_count,0) 
    as afternoon_leave_count, 
         --晚上出勤人数
        cad.evening_att_count as evening_att_count,
         --晚上请假人数
        if(cld.evening_leave_count is not null,cld.evening_leave_count,0) 
    as evening_leave_count,
        substring(css.studying_date,1,4) as yearinfo,
        substring(css.studying_date,6,2) as monthinfo,
        substring(css.studying_date,9,2) as dayinfo
from itcast_dws.class_attendance_dwm cad left join 
    itcast_dws.class_leave_dwm cld on cad.class_id=cld.class_id 
    left join itcast_ods.class_studying_student_count css 
    on cad.class_id=css.class_id and cad.dateinfo=css.studying_date
)
insert into itcast_dws.class_truant_dwm partition(yearinfo, monthinfo, dayinfo)
select 
    dateinfo,
    class_id,
    sum(studying_student_count - morning_att_count - morning_leave_count) 
    as morning_truant_count,
    sum(studying_student_count - afternoon_att_count - afternoon_leave_count) 
    as afternoon_truant_count,
    sum(studying_student_count - evening_att_count - evening_leave_count) 
    as evening_truant_count,
    yearinfo,
    monthinfo,
    dayinfo

from clc  where class_id is not null group by dateinfo,class_id,yearinfo,monthinfo,dayinfo ;
```
