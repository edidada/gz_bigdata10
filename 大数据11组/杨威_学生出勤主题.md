## 学生出勤表

```hive
-- 学生出勤信息表
CREATE TABLE IF NOT EXISTS edu_dws.student_attendance_dwm (
    dateinfo        String        comment '日期',
    class_id          int           comment '班级id',
    student_id        int           comment '学员id',
    morning_att       String       comment '上午出勤情况：0.正常出勤、1.迟到、2.其他（请假+旷课）',
    afternoon_att    String        comment '下午出勤情况：0.正常出勤、1.迟到、2.其他（请假+旷课）',
    evening_att       String        comment '晚自习出勤情况：0.正常出勤、1.迟到、2.其他（请假+旷课）')
comment '学生出勤(正常出勤和迟到)数据'
PARTITIONED BY (yearinfo STRING, monthinfo STRING, dayinfo STRING)
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY '\t'
stored as orcfile
TBLPROPERTIES ('orc.compress'='SNAPPY');



-- web_chat_ems
-- 动态分区配置
set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nonstrict;

-- hive压缩
set hive.exec.compress.intermediate=true;
set hive.exec.compress.output=true;
-- 写入时压缩生效
set hive.exec.orc.compression.strategy=COMPRESSION;
-- 写入数据强制分桶
set hive.enforce.bucketing=true;
-- 开启强制排序
set hive.enforce.sorting=true;


insert overwrite table itcast_dws.student_attendance_dwm partition(yearinfo,monthinfo,dayinfo)
select dtime.every_date as dateinfo,
       ctud.class_id as class_id,
       tssr.student_id as student_id,
       if(sum(
            case
            when substring(tssr.signin_time,12,8)  between from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') and tctt.morning_end_time
            then 1   --上午来了
            else 0 end   --上午没来
            ) > 0,  --打卡多次，只要有一次正常打卡，就会>0，返回true；否则没来，返回false
           if(
               sum(
                   case
               --上午打卡时间是否在上课前40分钟~上课后10分钟之内
                     when substring(tssr.signin_time,12,8)
                between from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') and from_unixtime(unix_timestamp(tctt.morning_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss') 
                      then 1    --正常出勤
                      else 0 end    --迟到
                ) > 0,              --有一次打卡是正常出勤，就会>0，返回true；否则迟到，返回false
               0,   --正常出勤
               1    --迟到
            ),
           2    --上午没来
        ) as morning_att,
       if(
           --下午正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
                   sum(case
                       when substring(tssr.signin_time,12,8) between from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss')  and tctt.afternoon_end_time
                               then 1
                           else 0 end) > 0,
                   if(sum(case
                          when substring(tssr.signin_time,12,8) between from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')-2400 ,'HH:mm:ss') and from_unixtime(unix_timestamp(tctt.afternoon_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss')
                               then 1
                              else 0 end) > 0, 0, 1), 2) as afternoon_att,
       if(
           --晚自习正常打卡为0，迟到10分钟以上为1，其他(请假+旷课)为2
                   sum(case
                       when substring(tssr.signin_time,12,8) between from_unixtime(unix_timestamp(tctt.evening_begin_time,'HH:mm:ss')-1200 ,'HH:mm:ss') and tctt.evening_end_time
                            then 1
                           else 0 end) > 0,
                   if(sum(case
                          when substring(tssr.signin_time,12,8) between from_unixtime(unix_timestamp(tctt.evening_begin_time,'HH:mm:ss')-1200 ,'HH:mm:ss') and from_unixtime(unix_timestamp(tctt.evening_begin_time,'HH:mm:ss')+600 ,'HH:mm:ss')
                              then 1
                              else 0 end) > 0, 0, 1), 2) as evening_att,
        substring(dtime.every_date, 1, 4) as yearinfo,
        substring(dtime.every_date, 6, 2) as monthinfo,
        substring(dtime.every_date, 6, 5) as dayinfo
    
                              
from (
         --获取今天之前一周内的日期
         select datelist as every_date from itcast_dwd.dim_calendar where datelist between '2019-09-01' and '2019-09-30'
     ) dtime
         --日期课表不为空且不是开班典礼
left join itcast_dwd.dim_course_table_upload_detail ctud
on ctud.class_date = dtime.every_date and coalesce(ctud.content, '') != '' and
ctud.content != '开班典礼'
    --学生打卡记录日期和班级匹配，且开启共屏进入学习
left join itcast_dwd.fact_tbh_student_signin_record tssr
on tssr.class_id = ctud.class_id and tssr.signin_date = dtime.every_date and tssr.share_state = 1
    --获取班级作息时间以判断是否按时出勤
left join itcast_dwd.dim_tbh_class_time_table tctt on tctt.id = tssr.time_table_id
     --按照日期、班级、学生分组统计
group by dtime.every_date, ctud.class_id, tssr.student_id,

        substring(dtime.every_date, 1, 4) ,
        substring(dtime.every_date, 6, 2) ,
        substring(dtime.every_date, 6, 5) ;
```





## 班级总人数出勤表

```hive
CREATE TABLE IF NOT EXISTS itcast_dws.class_attendance_dwm (
    dateinfo                 String          comment '日期',
    class_id                 int             comment '班级id',
    morning_att_count        String          comment '上午出勤人数',
    afternoon_att_count      String          comment '下午出勤人数',
    evening_att_count        String          comment '晚自习出勤人数',
    morning_late_count       String          comment '上午迟到人数',
    afternoon_late_count     String          comment '下午迟到人数',
    evening_late_count       String          comment '晚自习迟到人数')
comment '学生出勤(正常出勤和迟到)数据'
PARTITIONED BY (yearinfo STRING, monthinfo STRING, dayinfo STRING)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY '\t'
stored as orcfile
TBLPROPERTIES ('orc.compress'='SNAPPY');




insert overwrite table itcast_dws.class_attendance_dwm partition(yearinfo,monthinfo,dayinfo)
SELECT ctud.class_date ,
ctud.class_id,
--判断出勤是否有出勤=出勤或者迟到
count(distinct case when sad.morning_att = 0 or sad.morning_att = 1 THEN sad.student_id
else null
end
) as morning_att_count,
count(distinct case when sad.afternoon_att = 0 or sad.afternoon_att =1 THEN sad.student_id
else null
end
) as affternoon_att_count,
count(distinct case when sad.evening_att = 0 or sad.evening_att =1 THEN sad.student_id
else null
end
) as evening_att_count,
--判断条件=1为迟到，且数据不为空
count(distinct case when sad.morning_att = 1 and sad.morning_att IS NOT NULL THEN sad.student_id
else null
end
) as morning_late_count,
count(distinct case when sad.afternoon_att = 1 and sad.afternoon_att IS NOT NULL THEN sad.student_id
else null
end
) as afternoon_late_count,
count(distinct case when sad.evening_att = 1  and sad.evening_att IS NOT NULL THEN sad.student_id 
else null
end
) as evening_late_count,
substring(ctud.class_date,1,4) as yearinfo ,
substring(ctud.class_date,6,2) as monthinfo,
substring(ctud.class_date,6,5) as dayinfo
from itcast_dws.student_attendance_dwm sad LEFT JOIN itcast_dwd.dim_course_table_upload_detail ctud
on sad.class_id = ctud.class_id 
GROUP BY ctud.class_date,ctud.class_id,
substring(ctud.class_date,1,4) ,
substring(ctud.class_date,6,2) ,
substring(ctud.class_date,6,5) having ctud.class_id is not null;
```





## 班级请假表

```hive
CREATE TABLE IF NOT EXISTS edu_dws.class_leave_dwm (
    dateinfo                  String        comment '日期',
    class_id                  int           comment '班级id',
    morning_leave_count       String        comment '上午请假人数',
    afternoon_leave_count     String        comment '下午请假人数',
    evening_leave_count       String        comment '晚自习请假人数')
comment '班级请假数据统计'
PARTITIONED BY (yearinfo STRING, monthinfo STRING, dayinfo STRING)
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY '\t'
stored as orcfile
TBLPROPERTIES ('orc.compress'='SNAPPY');




insert overwrite table itcast_dws.class_leave_dwm partition(yearinfo,monthinfo,dayinfo)
select cud.class_date as dateinfo,
       cud.class_id,
--     关联判断请假周期，请假时间周期要与课程和作息时间对比
--      cud.class_date          课程表的上课日期     2020-09-16
--      ct.morning_begin_time   作息表的早上上课时间  09:00:00
--     请假结束时间 >= 2020-09-16 09:00:00 >= 请假开始时间，认为上午请假了
       count(distinct
       case when concat(cud.class_date, ' ', ct.morning_begin_time) >= sla.begin_time 
       and
       concat(cud.class_date, ' ', ct.morning_begin_time) <= sla.end_time
       then sla.student_id
       end)
    as morning_leave_count,
       count(distinct
       case when concat(cud.class_date, ' ', ct.afternoon_begin_time) >= sla.begin_time 
       and
       concat(cud.class_date, ' ', ct.afternoon_begin_time) <= sla.end_time
       then sla.student_id
       end)
    as afternoon_leave_count,
    count(distinct
       case when concat(cud.class_date, ' ', ct.evening_begin_time) >= sla.begin_time 
       and
       concat(cud.class_date, ' ', ct.evening_begin_time) <= sla.end_time
       then sla.student_id
       end)
    as evening_leave_count,
    substring(cud.class_date,1,4) ,
    substring(cud.class_date,6,2) ,
    substring(cud.class_date,6,5)

    
from itcast_dwd.fact_student_leave_apply as sla 
     inner join
     itcast_dwd.dim_tbh_class_time_table  as ct  on  sla.class_id = ct.class_id
     inner join
    itcast_dwd.dim_course_table_upload_detail  as cud on ct.class_id = cud.class_id

--       表关联
--   课程表，当天有课程内容
  AND cud.content IS NOT NULL
  AND cud.content != '开班典礼'
--   作息时间表，数据在生效期范围内
  where cud.class_date between ct.use_begin_date and ct.use_end_date
--   请假表，请假状态已审核通过，且没有取消、数据有效
  and sla.audit_state = 1
  and sla.cancel_state = 0
  and sla.valid_state = 1

group by cud.class_date, cud.class_id,
substring(cud.class_date,1,4) ,
substring(cud.class_date,6,2) ,
substring(cud.class_date,6,5);
```





## 班级旷课表

```hive
CREATE TABLE IF NOT EXISTS itcast_dws.class_truant_dwm (
    dateinfo                   String        comment '日期',
    class_id                   int           comment '班级id',
    morning_truant_count       String        comment '上午旷课人数',
    afternoon_truant_count     String        comment '下午旷课人数',
    evening_truant_count       String        comment '晚自习旷课人数')
comment '班级请假数据统计'
PARTITIONED BY (yearinfo STRING, monthinfo STRING, dayinfo STRING)
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY '\t'
stored as orcfile
TBLPROPERTIES ('orc.compress'='SNAPPY');




with class_count as (
    select
        ssc.studying_date as dateinfo,
        ssc.class_id,
        ssc.studying_student_count, -- 在读班级人数
        cad.morning_att_count,      -- 上午出勤人数
        cad.afternoon_att_count,    -- 中午出勤人数
        cad.evening_att_count,      -- 晚上出勤人数
        cld.morning_leave_count,    -- 上午请假人数
        cld.afternoon_leave_count,  -- 中午请假人数
        cld.evening_leave_count,    -- 晚上请假人数
        substring(cad.dateinfo,1,4 ) as yearinfo,
        substring(cad.dateinfo,6,2 ) as monthinfo,
        substring(cad.dateinfo,6,5) as dayinfo
        
    from itcast_dwd.dim_class_studying_student_count ssc
    join itcast_dws.class_attendance_dwm cad
    on ssc.class_id = cad.class_id and ssc.studying_date = cad.dateinfo
    left join itcast_dws.class_leave_dwm cld
    on ssc.class_id = cld.class_id and ssc.studying_date = cld.dateinfo
)
insert overwrite table itcast_dws.class_truant_dwm partition(yearinfo ,monthinfo ,dayinfo)
select
    dateinfo,
    class_id,
    -- 在读人数 - 上午出勤人数 - 上午请假人数 = 旷课人数
   cast(( studying_student_count - morning_att_count - (case when morning_leave_count is null then 0  else morning_leave_count end) ) as int)  as morning_truant_count,
    -- 在读人数 - 中午出勤人数 - 中午请假人数 = 旷课人数
   cast(( studying_student_count - afternoon_att_count - (case when afternoon_leave_count is null then 0 else afternoon_leave_count end) ) as int)   as afternoon_truant_count,
    -- 在读人数 - 晚上出勤人数 - 晚上请假人数 = 旷课人数
   cast(( studying_student_count - evening_att_count - (case when evening_leave_count is null then 0 else evening_leave_count end) ) as int)  as evening_truant_count,
    yearinfo,
    monthinfo,
    dayinfo
from class_count;
```

 )
